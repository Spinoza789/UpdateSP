	"txid" text PRIMARY KEY NOT NULL,
	"refund_id" text NOT NULL,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL
);
CREATE TABLE "group_buy_refund_attempts" (
	"id" text PRIMARY KEY NOT NULL,
	"refund_id" text NOT NULL,
	"submitted_txid" text NOT NULL,
	"processor_username" text NOT NULL,
	"processor_role" text NOT NULL,
	"started_at" timestamp with time zone DEFAULT now() NOT NULL,
	"ended_at" timestamp with time zone,
	"result_code" text,
	"failure_reason" text,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL
);
CREATE TABLE "group_buy_refund_notifications" (
	"id" text PRIMARY KEY NOT NULL,
	"refund_id" text NOT NULL,
	"account_id" text NOT NULL,
	"status" text NOT NULL,
	"event_key" text NOT NULL,
	"transition_key" text NOT NULL,
	"attempt_id" text,
	"order_code" text,
	"amount_usd" numeric(10, 2),
	"explorer_url" text,
	"txid" text,
	"attempt_count" integer DEFAULT 0 NOT NULL,
	"claim_token" text,
	"claimed_at" timestamp with time zone,
	"sent_at" timestamp with time zone,
	"last_error" text,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL,
	CONSTRAINT "group_buy_refund_notifications_status_check" CHECK (status = ANY (ARRAY['requested'::text, 'verifying'::text, 'verification_failed'::text, 'completed'::text]))
);
CREATE TABLE "group_buy_refunds" (
	"id" text PRIMARY KEY NOT NULL,
	"order_id" text NOT NULL,
	"group_buy_id" text NOT NULL,
	"account_id" text NOT NULL,
	"method" text NOT NULL,
	"status" text DEFAULT 'requested' NOT NULL,
	"amount_usd" numeric(10, 2) NOT NULL,
	"asset" text,
	"network" text,
	"destination_wallet" text,
	"amount_base_units" numeric(30, 0),
	"outgoing_txid" text,
	"confirmation_count" integer,
	"explorer_url" text,
	"processor_username" text,
	"processor_role" text,
	"processing_attempt_id" text,
	"verification_error_code" text,
	"verification_error_message" text,
	"requested_at" timestamp with time zone DEFAULT now() NOT NULL,
	"verifying_at" timestamp with time zone,
	"failed_at" timestamp with time zone,
	"completed_at" timestamp with time zone,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL,
	"updated_at" timestamp with time zone DEFAULT now() NOT NULL,
	CONSTRAINT "group_buy_refunds_asset_check" CHECK ((asset IS NULL) OR (asset = ANY (ARRAY['USDC'::text, 'USDT'::text]))),
	CONSTRAINT "group_buy_refunds_method_check" CHECK (method = ANY (ARRAY['credit'::text, 'crypto'::text])),
	CONSTRAINT "group_buy_refunds_method_fields_check" CHECK (((method = 'credit'::text) AND (asset IS NULL) AND (network IS NULL) AND (destination_wallet IS NULL) AND (amount_base_units IS NULL)) OR ((method = 'crypto'::text) AND (asset IS NOT NULL) AND (network IS NOT NULL) AND (destination_wallet IS NOT NULL) AND (amount_base_units IS NOT NULL) AND (amount_base_units > (0)::numeric))),
	CONSTRAINT "group_buy_refunds_network_check" CHECK ((network IS NULL) OR (network = 'Ethereum ERC-20'::text)),
	CONSTRAINT "group_buy_refunds_positive_amount_check" CHECK (amount_usd > (0)::numeric),
	CONSTRAINT "group_buy_refunds_status_check" CHECK (status = ANY (ARRAY['requested'::text, 'verifying'::text, 'verification_failed'::text, 'completed'::text]))
);
CREATE TABLE "group_buy_refund_transition_outbox" (
	"id" text PRIMARY KEY NOT NULL,
	"refund_id" text NOT NULL,
	"attempt_id" text NOT NULL,
	"order_id" text,
	"group_buy_id" text,
	"account_id" text,
	"method" text,
	"source" text DEFAULT 'background' NOT NULL,
	"transition" text NOT NULL,
	"action" text DEFAULT 'group_buy_refund_transition' NOT NULL,
	"processor_username" text NOT NULL,
	"processor_role" text NOT NULL,
	"submitted_txid" text,
	"result_code" text NOT NULL,
	"delivered_at" timestamp with time zone,
	"attempt_count" integer DEFAULT 0 NOT NULL,
	"last_attempt_at" timestamp with time zone,
	"last_error" text,
	"created_at" timestamp with time zone DEFAULT now() NOT NULL,
	CONSTRAINT "group_buy_refund_transition_outbox_txid_nonempty" CHECK ((submitted_txid IS NULL) OR (length(TRIM(BOTH FROM submitted_txid)) > 0))
);
