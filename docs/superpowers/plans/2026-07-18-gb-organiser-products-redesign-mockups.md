# GB Organiser V2 Products Redesign Mockups Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build five standalone, interactive desktop Products mockups in the mockup sandbox using the approved GB Organiser V2 shell and a shared deterministic 120-product fixture, without modifying production organiser code.

**Architecture:** A private underscore-prefixed directory contains the shell, pure catalogue model helpers, shared product form/import/confirmation controls, and feedback primitives. The mockup preview plugin ignores underscore-prefixed paths, so only the five concept entry components become direct preview routes. Each concept owns its local state and composition while consuming the same typed fixture and shared controls.

**Tech Stack:** React 19, TypeScript, Vite mockup preview plugin, scoped CSS, Lucide React, Node test runner with --experimental-strip-types, and existing sandbox dependencies only.

import test from "node:test";
import { SAMPLE_PRODUCTS } from "../data";
import {
  applyBulkPatch,

  filterProducts,
  getProductStatus,
  resetProducts,

test("fixture is deterministic and supports high-volume comparison", () => {
  assert.equal(SAMPLE_PRODUCTS.length, 120);
  assert.deepEqual(SAMPLE_PRODUCTS.slice(0, 3).map(p => p.id), ["prod-001", "prod-002", "prod-003"]);
  assert.ok(SAMPLE_PRODUCTS.some(p => p.stock === null));
  assert.ok(SAMPLE_PRODUCTS.some(p => p.stock === 0));
  assert.ok(SAMPLE_PRODUCTS.some(p => p.maxPerCustomer !== null));
  assert.ok(SAMPLE_PRODUCTS.some(p => p.halfKitEnabled));

test("filterProducts searches name, vendor, size, and category", () => {
  const result = filterProducts(SAMPLE_PRODUCTS, { query: "qsc", vendor: "all", category: "all", stock: "all" });
  assert.ok(result.length > 0);
  assert.ok(result.every(p => p.vendor.toLowerCase().includes("qsc")));
});

test("bulk patch is immutable and reset restores the fixture", () => {
  const ids = SAMPLE_PRODUCTS.slice(0, 2).map(p => p.id);
  const changed = applyBulkPatch(SAMPLE_PRODUCTS, ids, { price: 49, stock: 10 });
  assert.equal(changed[0].price, 49);
  assert.equal(changed[1].stock, 10);
  assert.equal(changed[2].price, SAMPLE_PRODUCTS[2].price);
  assert.notEqual(changed, SAMPLE_PRODUCTS);
  assert.deepEqual(resetProducts(changed), SAMPLE_PRODUCTS);
});

test("status and import classification expose reviewable states", () => {
  assert.equal(getProductStatus({ ...SAMPLE_PRODUCTS[0], stock: 0 }), "out-of-stock");
  assert.equal(getProductStatus({ ...SAMPLE_PRODUCTS[0], stock: 4 }), "low-stock");
  const rows = classifyImportRows(SAMPLE_PRODUCTS, [
    { name: SAMPLE_PRODUCTS[0].name, vendor: SAMPLE_PRODUCTS[0].vendor, mgSize: SAMPLE_PRODUCTS[0].mgSize, price: SAMPLE_PRODUCTS[0].price },
    { name: "New peptide", vendor: "QSC", mgSize: "5mg", price: 29 },
  ]);
  assert.equal(rows[0].status, "duplicate");
  assert.equal(rows[1].status, "new");
});
